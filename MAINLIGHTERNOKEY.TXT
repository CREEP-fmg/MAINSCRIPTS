local function _e(_f)
    if type(_f) ~= "string" or #_f == 0 then
        return nil
    end

    if type(buffer) == "table"
        and type(buffer.frombase64) == "function"
        and type(buffer.tostring) == "function" then

        local _g, _h = pcall(function()
            return buffer.tostring(
                buffer.frombase64(_f)
            )
        end)

        if _g and type(_h) == "string" then
            return _h
        end
    end

    if type(crypt) == "table"
        and type(crypt.base64decode) == "function" then

        local _i, _j = pcall(
            crypt.base64decode,
            _f
        )

        if _i and type(_j) == "string" then
            return _j
        end
    end

    if type(base64) == "table"
        and type(base64.decode) == "function" then

        local _k, _l = pcall(
            base64.decode,
            _f
        )

        if _k and type(_l) == "string" then
            return _l
        end
    end

    local _m =
        "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
        .. "abcdefghijklmnopqrstuvwxyz"
        .. "0123456789+/"

    local _n = {}

    for _o = 1, #_m do
        _n[string.byte(_m, _o)] = _o - 1
    end

    _f = string.gsub(
        _f,
        "[^" .. _m .. "=]",
        ""
    )

    local _p = {}

    for _q = 1, #_f, 4 do
        local _r, _s, _t, _u =
            string.byte(
                _f,
                _q,
                _q + 3
            )

        if not _r or not _s then
            break
        end

        local _v = _n[_r]
        local _w = _n[_s]

        if _v == nil or _w == nil then
            break
        end

        local _x = 0
        local _y = 0

        if _t and _t ~= 61 then
            _x = _n[_t] or 0
        end

        if _u and _u ~= 61 then
            _y = _n[_u] or 0
        end

        local _z

        if bit32 then
            _z = bit32.bor(
                bit32.lshift(_v, 2),
                bit32.rshift(_w, 4)
            )
        else
            _z =
                _v * 4 +
                math.floor(_w / 16)
        end

        _p[#_p + 1] =
            string.char(_z)

        if _t and _t ~= 61 then
            local _aa

            if bit32 then
                _aa = bit32.bor(
                    bit32.lshift(
                        bit32.band(_w, 15),
                        4
                    ),
                    bit32.rshift(_x, 2)
                )
            else
                _aa =
                    (_w % 16) * 16 +
                    math.floor(_x / 4)
            end

            _p[#_p + 1] =
                string.char(_aa)
        end

        if _u and _u ~= 61 then
            local _ab

            if bit32 then
                _ab = bit32.bor(
                    bit32.lshift(
                        bit32.band(_x, 3),
                        6
                    ),
                    _y
                )
            else
                _ab =
                    (_x % 4) * 64 +
                    _y
            end

            _p[#_p + 1] =
                string.char(_ab)
        end
    end

    local _ac = table.concat(_p)

    if #_ac == 0 then
        return nil
    end

    return _ac
end

local function _ad()
    task.spawn(function()

        local _ae, _af = pcall(function()
            return game:HttpGet(
                _c("MAINLIGHTER.TXT")
            )
        end)

        if not _ae
            or type(_af) ~= "string"
            or #_af == 0 then

            warn("[ML] Не удалось получить данные.")
            return
        end

        local _ag, _ah = pcall(
            _e,
            _af
        )

        if not _ag
            or type(_ah) ~= "string"
            or #_ah == 0 then

            warn("[ML] Некорректные данные.")
            return
        end

        if type(loadstring) ~= "function" then
            warn("[ML] loadstring недоступен.")
            return
        end

        local _ai, _aj =
            loadstring(_ah)

        if not _ai then
            warn(
                "[ML] Ошибка компиляции: "
                .. tostring(_aj)
            )
            return
        end

        local _ak, _al =
            pcall(_ai)

        if not _ak then
            warn(
                "[ML] Ошибка выполнения: "
                .. tostring(_al)
            )
        end
    end)
end
