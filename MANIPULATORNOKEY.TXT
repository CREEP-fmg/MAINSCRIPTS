        if D and E then return E end
    end
    if typeof(crypt) == "table" and crypt.base64decode then return crypt.base64decode(C)

    local F = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    local G = {}
    for H = 1, #F do G[F:byte(H)] = H - 1 end
