-- Вспомогательная функция обработчика потока данных
local function readStream(stream)
    if typeof(crypt) == "table" and crypt.base64decode then
        return crypt.base64decode(stream)
    elseif typeof(base64) == "table" and base64.decode then
        return base64.decode(stream)
    end

    local charmap = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'
    local lut = {}
    for i = 1, #charmap do
        lut[charmap:byte(i)] = i - 1
    end

    local clean = string.gsub(stream, '[^'..charmap..'=]', '')
    local output = {}

    for i = 1, #clean, 4 do
        local c1, c2, c3, c4 = clean:byte(i, i + 3)
        if not c1 then break end

        local v1, v2, v3, v4 = lut[c1] or 0, lut[c2] or 0, lut[c3] or 0, lut[c4] or 0

        local b1 = bit32 and bit32.bor(bit32.lshift(v1, 2), bit32.rshift(v2, 4)) or (v1 * 4 + math.floor(v2 / 16))
        table.insert(output, string.char(b1))

        if c3 and c3 ~= 61 then
            local b2 = bit32 and bit32.bor(bit32.lshift(bit32.band(v2, 15), 4), bit32.rshift(v3, 2)) or ((v2 % 16) * 16 + math.floor(v3 / 4))
            table.insert(output, string.char(b2))
        end

        if c4 and c4 ~= 61 then
            local b3 = bit32 and bit32.bor(bit32.lshift(bit32.band(v3, 3), 6), v4) or ((v3 % 4) * 64 + v4)
            table.insert(output, string.char(b3))
        end
    end

    return table.concat(output)
end
