local function read(stream)
    local charmap = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'
    local clean = string.gsub(stream, '[^'..charmap..'=]', '')
    return (clean:gsub('.', function(char)
        if (char == '=') then return '' end
        local bitstr, idx = '', (charmap:find(char) - 1)
        for i = 6, 1, -1 do 
            bitstr = bitstr .. (idx % 2^i - idx % 2^(i-1) >= 2^(i-1) and '1' or '0') 
        end
        return bitstr
    end):gsub('%d%d%d%d%d%d%d%d', function(byte)
        local val = 0
        for i = 1, 8 do 
            val = val + (byte:sub(i, i) == '1' and 2^(8 - i) or 0) 
        end
        return string.char(val)
    end))
end
